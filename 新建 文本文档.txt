  <script>
    (() => {
      const DATA_KEY = 'recitation.offline.data.v1';
      const DEFAULTS = { dailyGoal: 108, reminderEnabled: false, startDate: '' };
      const today = dateKey(new Date());
      let activePeriod = 'day';
      let toastTimer;
      let storageAvailable = true;
      let state = loadState();

      function dateKey(date) {
        const year = date.getFullYear();
        const month = String(date.getMonth() + 1).padStart(2, '0');
        const day = String(date.getDate()).padStart(2, '0');
        return `${year}-${month}-${day}`;
      }

      function parseDate(key) {
        const [year, month, day] = key.split('-').map(Number);
        return new Date(year, month - 1, day, 12);
      }

      function loadState() {
        try {
          const saved = JSON.parse(localStorage.getItem(DATA_KEY) || '{}');
          return {
            records: saved.records && typeof saved.records === 'object' ? saved.records : {},
            settings: { ...DEFAULTS, ...(saved.settings || {}) }
          };
        } catch (error) {
          storageAvailable = false;
          return { records: {}, settings: { ...DEFAULTS } };
        }
      }

      function saveState() {
        try {
          localStorage.setItem(DATA_KEY, JSON.stringify(state));
          storageAvailable = true;
        } catch (error) {
          storageAvailable = false;
        }
        document.getElementById('storage-warning').style.display = storageAvailable ? 'none' : 'block';
      }

      function showToast(message) {
        const toast = document.getElementById('toast');
        toast.textContent = message;
        toast.classList.add('visible');
        clearTimeout(toastTimer);
        toastTimer = setTimeout(() => toast.classList.remove('visible'), 2300);
      }

      function getRecord(date) {
        return state.records[date] || { count: 0, makeupCount: 0, goal: state.settings.dailyGoal };
      }

      function addCount(amount, isMakeup = false) {
        const record = getRecord(today);
        record.count += amount;
        if (isMakeup) record.makeupCount += amount;
        state.records[today] = record;
        saveState();
        render();
        showToast(isMakeup ? `已登记补念 ${amount} 次` : `已记入 ${amount} 次`);
      }

      function dayRange(start, end) {
        const result = [];
        const cursor = parseDate(start);
        const last = parseDate(end);
        while (cursor <= last) {
          result.push(dateKey(cursor));
          cursor.setDate(cursor.getDate() + 1);
        }
        return result;
      }

      function calculateDebt() {
        const dates = Object.keys(state.records).sort();
        const start = state.settings.startDate || dates[0] || today;
        const outstanding = [];
        for (const date of dayRange(start, today)) {
          const record = getRecord(date);
          let makeup = record.makeupCount || 0;
          for (const debt of outstanding) {
            const applied = Math.min(debt.remaining, makeup);
            debt.remaining -= applied;
            makeup -= applied;
            if (makeup === 0) break;
          }
          if (date < today) {
            const goal = record.goal || state.settings.dailyGoal;
            const deficit = Math.max(0, goal - (record.count || 0));
            if (deficit > 0) outstanding.push({ remaining: deficit });
          }
        }
        return outstanding.reduce((sum, debt) => sum + debt.remaining, 0);
      }

      function renderHome() {
        const record = getRecord(today);
        const goal = state.settings.dailyGoal;
        const remaining = Math.max(0, goal - record.count);
        document.getElementById('today-count').textContent = record.count;
        document.getElementById('goal-caption').textContent = `每日目标 ${goal} 次`;
        document.getElementById('remaining-caption').textContent = remaining ? `还差 ${remaining} 次` : '今日目标已完成';
        document.getElementById('progress-fill').style.width = `${Math.min(100, Math.round(record.count / goal * 100))}%`;
        document.getElementById('debt-count').textContent = calculateDebt();
      }

      function periodRange(mode, selected) {
        const date = parseDate(selected);
        const year = date.getFullYear();
        const month = date.getMonth();
        if (mode === 'day') return [selected, selected];
        if (mode === 'week') {
          const start = new Date(date);
          start.setDate(start.getDate() - ((start.getDay() + 6) % 7));
          const end = new Date(start);
          end.setDate(end.getDate() + 6);
          return [dateKey(start), dateKey(end)];
        }
        if (mode === 'month') return [dateKey(new Date(year, month, 1, 12)), dateKey(new Date(year, month + 1, 0, 12))];
        return [`${year}-01-01`, `${year}-12-31`];
      }

      function renderStats() {
        const selected = document.getElementById('stats-date').value || today;
        const [start, end] = periodRange(activePeriod, selected);
        const lastDate = end > today ? today : end;
        const dates = start <= lastDate ? dayRange(start, lastDate) : [];
        const total = dates.reduce((sum, date) => sum + (getRecord(date).count || 0), 0);
        const names = { day: '当日', week: '本周', month: '本月', year: '本年' };
        const label = activePeriod === 'day' ? start : activePeriod === 'week' ? `${start} 至 ${end}` : activePeriod === 'month' ? start.slice(0, 7) : start.slice(0, 4);
        document.getElementById('summary-period').textContent = `${names[activePeriod]}累计 · ${label}`;
        document.getElementById('period-total').textContent = total;
        document.getElementById('day-count-label').textContent = `${dates.length} 天`;
        const list = document.getElementById('day-list');
        if (!dates.length) {
          list.innerHTML = '<div class="empty">这个时段还没有记录</div>';
          return;
        }
        list.innerHTML = dates.reverse().map((date) => {
          const record = getRecord(date);
          const makeup = record.makeupCount ? ` · 补念 ${record.makeupCount} 次` : '';
          return `<div class="day-row"><div><div class="day-date">${date}</div><div class="day-meta">目标 ${record.goal || state.settings.dailyGoal} 次${makeup}</div></div><div class="day-count">${record.count || 0}<small> 次</small></div></div>`;
        }).join('');
      }

      function renderSettings() {
        document.getElementById('daily-goal').value = state.settings.dailyGoal;
        document.getElementById('reminder-enabled').checked = state.settings.reminderEnabled;
        document.getElementById('storage-warning').style.display = storageAvailable ? 'none' : 'block';
      }

      function validateAmount(value) {
        const amount = Number(value);
        if (!Number.isInteger(amount) || amount < 1 || amount > 999999) {
          showToast('请输入 1 至 999999 的整数');
          return null;
        }
        return amount;
      }

      function normalizeSpeech(text) {
        return text.toLocaleLowerCase().replace(/[\s，。！？、；：,.!?;:…~"'“”‘’（）()【】\[\]《》]/g, '');
      }

      function countPhraseOccurrences(transcript, phrase) {
        const text = normalizeSpeech(transcript);
        const target = normalizeSpeech(phrase);
        if (!target) return 0;
        let count = 0;
        let position = 0;
        while ((position = text.indexOf(target, position)) !== -1) {
          count += 1;
          position += target.length;
        }
        return count;
      }

      let recognition = null;
      let listening = false;
      let lastHandledResult = 0;
      let voiceRestartTimer = null;
      let voiceSessionCount = 0;

      function setVoiceStatus(message, active = listening) {
        document.getElementById('voice-status-text').textContent = message;
        document.getElementById('voice-status').dataset.listening = String(active);
        document.getElementById('voice-toggle').setAttribute('aria-pressed', String(active));
        document.getElementById('voice-toggle').textContent = active ? '停止识别' : '开始识别';
      }

      function stopVoiceRecognition() {
        listening = false;
        clearTimeout(voiceRestartTimer);
        if (recognition) {
          try { recognition.stop(); } catch (error) { setVoiceStatus('已停止', false); }
        } else {
          setVoiceStatus('已停止', false);
        }
      }

      function startVoiceRecognition() {
        const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
        if (!SpeechRecognition) {
          setVoiceStatus('此浏览器不支持语音识别，请换用支持语音识别的浏览器', false);
          return;
        }

        const phrase = document.getElementById('voice-phrase').value.trim();
        if (!normalizeSpeech(phrase)) {
          showToast('请先填写要计数的短句');
          document.getElementById('voice-phrase').focus();
          return;
        }

        state.settings.voicePhrase = phrase;
        saveState();
        voiceSessionCount = 0;
        lastHandledResult = 0;
        document.getElementById('voice-session-count').textContent = '0';
        document.getElementById('voice-transcript').textContent = '正在等待语音…';

        try {
          recognition = new SpeechRecognition();
          recognition.lang = 'zh-CN';
          recognition.continuous = true;
          recognition.interimResults = true;
          recognition.maxAlternatives = 1;
          recognition.onresult = (event) => {
            let latestTranscript = '';
            for (let index = event.resultIndex; index < event.results.length; index += 1) {
              const result = event.results[index];
              latestTranscript += result[0].transcript;
              if (!result.isFinal || index < lastHandledResult) continue;
              lastHandledResult = index + 1;
              const occurrences = countPhraseOccurrences(result[0].transcript, phrase);
              if (occurrences > 0) {
                voiceSessionCount += occurrences;
                document.getElementById('voice-session-count').textContent = voiceSessionCount;
                addCount(occurrences);
              }
            }
            if (latestTranscript) document.getElementById('voice-transcript').textContent = latestTranscript;
          };
          recognition.onerror = (event) => {
            if (event.error === 'not-allowed' || event.error === 'service-not-allowed') {
              listening = false;
              setVoiceStatus('麦克风或语音识别权限未允许', false);
            } else if (event.error === 'no-speech') {
              setVoiceStatus('没有听清，继续聆听…', true);
            } else if (event.error !== 'aborted') {
              listening = false;
              setVoiceStatus(`识别暂不可用：${event.error}`, false);
            }
          };
          recognition.onend = () => {
            if (listening) {
              voiceRestartTimer = setTimeout(() => {
                if (!listening) return;
                try { recognition.start(); } catch (error) {
                  listening = false;
                  setVoiceStatus('识别已暂停，请重新开始', false);
                }
              }, 250);
            } else {
              setVoiceStatus('已停止', false);
            }
          };
          listening = true;
          setVoiceStatus('正在聆听…', true);
          recognition.start();
        } catch (error) {
          listening = false;
          setVoiceStatus('无法启动麦克风，请检查浏览器权限', false);
        }
      }

      function exportBackup() {
        const blob = new Blob([JSON.stringify({ version: 1, ...state }, null, 2)], { type: 'application/json' });
        const url = URL.createObjectURL(blob);
        const link = document.createElement('a');
        link.href = url;
        link.download = `念经积累备份-${today}.json`;
        link.click();
        URL.revokeObjectURL(url);
        showToast('备份文件已生成');
      }

      function importBackup(file) {
        const reader = new FileReader();
        reader.onload = () => {
          try {
            const data = JSON.parse(reader.result);
            if (!data || !data.records || typeof data.records !== 'object' || !data.settings || typeof data.settings !== 'object') throw new Error('invalid backup');
            state = { records: data.records, settings: { ...DEFAULTS, ...data.settings } };
            saveState();
            render();
            showToast('备份已导入');
          } catch (error) {
            showToast('备份文件无效，未导入');
          }
          document.getElementById('backup-file').value = '';
        };
        reader.readAsText(file);
      }
